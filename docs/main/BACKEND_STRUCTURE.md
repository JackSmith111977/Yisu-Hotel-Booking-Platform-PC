### 4.2 服务端客户端
```typescript
// src/lib/supabase_server.ts
import { createServerClient } from '@supabase/ssr'
import { cookies } from 'next/headers'

export async function createClient() {
  const cookieStore = await cookies()

  return createServerClient(
    process.env.NEXT_PUBLIC_SUPABASE_URL!,
    process.env.NEXT_PUBLIC_SUPABASE_ANON_KEY!,
    {
      cookies: {
        getAll() {
          return cookieStore.getAll()
        },
        setAll(cookiesToSet) {
          try {
            cookiesToSet.forEach(({ name, value, options }) =>
              cookieStore.set(name, value, options)
            )
          } catch {
            // 在 Server Components 中忽略
          }
        },
      },
    }
  )
}
```

## 5. 行级安全策略 (RLS)

### 5.1 hotels 表策略
```sql
-- 商户只能查看自己的酒店
CREATE POLICY "商户查看自己的酒店" ON hotels
  FOR SELECT USING (auth.uid() = merchant_id);

-- 商户可以创建自己的酒店
CREATE POLICY "商户创建酒店" ON hotels
  FOR INSERT WITH CHECK (auth.uid() = merchant_id);

-- 商户可以更新自己的酒店
CREATE POLICY "商户更新自己的酒店" ON hotels
  FOR UPDATE USING (auth.uid() = merchant_id);

-- 管理员可以查看所有酒店
CREATE POLICY "管理员查看所有酒店" ON hotels
  FOR SELECT USING (
    EXISTS (
      SELECT 1 FROM users
      WHERE users.id = auth.uid() AND users.role = 'admin'
    )
  );
```

### 5.2 users 表策略

## 6. 错误处理

### 6.1 统一错误格式
```typescript
// ✅ 推荐：统一返回格式
type ActionResult<T> =
  | { success: true; data: T }
  | { success: false; error: string }

export async function someAction(): Promise<ActionResult<SomeData>> {
  try {
    // ... 业务逻辑
    return { success: true, data: result }
  } catch (error) {

  }

```

### 6.2 常见错误码
| 错误类型 | 错误信息 | 处理建议 |
|----------|----------|----------|
| `AUTH_REQUIRED` | 请先登录 | 重定向到登录页 |
| `PERMISSION_DENIED` | 无权限操作 | 显示权限提示 |
| `NOT_FOUND` | 资源不存在 | 显示 404 页面 |
| `VALIDATION_ERROR` | 数据验证失败 | 显示表单错误 |
| `NETWORK_ERROR` | 网络错误 | 重试或提示用户 |
